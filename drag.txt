    <script>
        $(document).ready(function () {
    var map = L.map('map').setView([33.8869, 9.5375], 7);

    L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', {
        maxZoom: 18,
        attribution: 'Map data &copy; <a href="https://www.openstreetmap.org/copyright">OpenStreetMap</a> contributors'
    }).addTo(map);

    var houseIcon = L.icon({
        iconUrl: '/house-icon.png',
        iconSize: [32, 32],
        iconAnchor: [16, 32],
        popupAnchor: [0, -32]
    });

    var draggableEntities = document.querySelectorAll('.draggable');
    var formContainer = document.getElementById('form-container');
    var entityForm = document.getElementById('entity-form');
    var currentMarker = null;

   
            var lineDrawingMode = false;
            var selectedPoints = [];

            draggableEntities.forEach(function (entity) {
                entity.addEventListener('click', function (event) {
                    var entityId = event.currentTarget.id;
                    if (entityId === 'draggable-line') {
                        lineDrawingMode = true;
                        selectedPoints = [];
                        map.once('click', function (e) {
                            if (lineDrawingMode) {
                                selectedPoints.push([e.latlng.lat, e.latlng.lng]);

                                if (selectedPoints.length === 2) {
                                    var line = L.polyline(selectedPoints, { color: 'blue' }).addTo(map);

                                    // Reset drawing mode
                                    lineDrawingMode = false;
                                }
                            }
                        });
                    } else if (entityId === 'draggable-entity') {
                        map.once('click', function (e) {
                            var offset = e.latlng;
                            L.marker([offset.lat, offset.lng], { icon: houseIcon }).addTo(map)
                                .bindPopup('Entity at ' + offset.lat + ', ' + offset.lng)
                                .openPopup();
                        });
                    }
                });
            });
    
  

    map.getContainer().addEventListener('dragover', function (event) {
        event.preventDefault();
        event.dataTransfer.dropEffect = 'move';
    });

    map.getContainer().addEventListener('drop', function (event) {
        event.preventDefault();
        var entityId = event.dataTransfer.getData('text/plain');
        var entityName = document.getElementById(entityId).textContent;

        if (entityId === 'draggable-entity') {
            var offset = map.containerPointToLatLng(L.point(event.clientX, event.clientY));

            currentMarker = L.marker([offset.lat, offset.lng], { icon: houseIcon }).addTo(map);
            currentMarker.bindPopup(entityName + ' at ' + offset.lat + ', ' + offset.lng).openPopup();

            var geoJSON = {
                type: "Point",
                coordinates: [offset.lng, offset.lat]
            };

            formContainer.style.display = 'block';
            formContainer.style.left = event.clientX + 'px';
            formContainer.style.top = event.clientY + 'px';

            document.getElementById('entity-geom').value = JSON.stringify(geoJSON);
        }
    });

    $(entityForm).submit(function (e) {
        e.preventDefault();

        $.post('/save-entity', $(this).serialize(), function (response) {
            console.log('Entity saved successfully:', response);
            formContainer.style.display = 'none';
        }).fail(function (jqXHR, textStatus, errorThrown) {
            console.error('Error saving entity:', textStatus, errorThrown);
            if (currentMarker) {
                map.removeLayer(currentMarker);
                currentMarker = null;
            }
        });
    });

    // Fetch points from the server and add them to the map
    $.getJSON("{{ path('fetch_data') }}", function (data) {
        if (Array.isArray(data)) {
            data.forEach(function (entity) {
                var geometry = JSON.parse(entity.geometry);
                if (geometry.type === "Point") {
                    var marker = L.marker([geometry.coordinates[1], geometry.coordinates[0]], { icon: houseIcon })
                        .addTo(map)
                        .bindPopup(entity.reference + '<br>' + JSON.stringify(entity.infos));
                }
            });
        } else {
            console.error("Unexpected data format:", data);
        }
    }).fail(function (jqXHR, textStatus, errorThrown) {
        console.error("Error fetching data:", textStatus, errorThrown);
    });
});

  
    </script>
